    # Use regex to find the USER_COMPILE_DEFINITIONS block and replace the empty "" with our flag
    # This safely preserves the formatting and parentheses exactly as Vitis expects
    updated_content = re.sub(
        r'(set\(USER_COMPILE_DEFINITIONS\s*)""(\s*\))',
        r'\1"USE_SHEF_FIXED_CONFIG"\2',
        content
    )
    
    # Write the modified content back to the file
    with open(fsbl_cmake_path, 'w') as f:
        f.write(updated_content)
    
    print("Successfully injected USE_SHEF_FIXED_CONFIG into the FSBL UserConfig.cmake")

import subprocess

print("Preparing to generate BOOT.BIN...")

# 1. Define the BIF content dynamically using our exact relative paths
# Stepping up into embedded_shef to grab the compiled .elf files
bif_content = """//arch = zynqmp; split = false; format = BIN
the_ROM_image:
{
    [bootloader, destination_cpu = a53-0] ../embedded_shef/platform/zynqmp_fsbl/build/fsbl.elf
    [destination_cpu = pmu] ../embedded_shef/shef_pmufw/build/shef_pmufw.elf
    [destination_cpu = r5-0] ../embedded_shef/security_kernel/build/security_kernel.elf
    [destination_cpu = a53-0] ../embedded_shef/runtime/build/runtime.elf
}
"""
bif_filename = "package.bif"

    f.write(bif_content)
